%% This exists to keep the two shared style files honest. A fragment that
%% compiles here depends on nothing outside them, which is the property
%% scripts/gen_interface.py relies on when it reads the same fragment: the
%% generator expands macros it finds in ucgamers.sty and in
%% functionalities/encyclopedia.sty and nowhere else, so a fragment that needs
%% something from main.tex's own preamble would render in the book and break
%% silently on the web. Load the same two here, in the same order, or this
%% file stops being that check.

\usepackage[most]{tcolorbox}

\usepackage{ucgamers}
%% \input rather than \usepackage: it lives in functionalities/ rather than on
%% TeX's search path, deliberately -- it is not an input of the book, and
%% scripts/artifact_manifest.py watches latex/*.sty.
