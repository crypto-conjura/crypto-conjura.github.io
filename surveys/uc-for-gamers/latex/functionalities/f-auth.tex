%% The interface box of f-auth, the one source for both the book
%% and uc/.../f-auth/index.qmd. Edit here and nowhere else:
%% scripts/gen_interface.py regenerates the page's HTML from this file,
%% and CI fails the build when the two have drifted apart.
\begin{interface}[nofloat]{Functionality $\Fauth$ \\
\params{$\PID$, \quad $\Ps$, \quad $\admits$, \quad
$\uses := \{(\Adv,\serves)\}$, \quad $\pars := \none$}}

{\scriptsize
\setlength{\columnsep}{1.4em}
\begin{multicols}{2}
\opsig{$\op{Initialize}()$}:
\begin{algorithmic}[1]
\setcounter{contline}{0}\algcont
\State $\V{A} \gets \unset$; \ $\V{B} \gets \unset$
  \Comment{sender, intended receiver}
\State $\V{m} \gets \unset$
\State $\V{corr} \gets 0$; \ $\V{done} \gets 0$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Send}(B, \msg)$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\Req{$\V{A} = \unset \ \wedge\ \msg \in \Msgs$} \label{ln:authonce}
  \Comment{one message to an instance}
\State $\V{A} \gets \id.P$; \ $\V{B} \gets B$; \ $\V{m} \gets \msg$
\State $\Adv\bigl(\id.\op{Send},\ \V{A},\ \V{B},\ \V{m}\bigr)$ \label{ln:authleak}
  \Comment{in the clear: authenticity, not secrecy}
\State \textbf{return} $\ok$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Report}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\Req{$\id.P = \V{A}$}
\State \textbf{return} $\V{corr}$ \label{ln:authreport}
  \Comment{the sender may ask, afterwards}
\algsave
\end{algorithmic}
\vspace{1.4ex}

\columnbreak

\opsig{$\id.\op{Fetch}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\Req{$\V{A} \neq \unset \ \wedge\ \V{done} = 0$}
\State $a \gets \Adv^{!}\bigl(\id.\op{Fetch},\ \id.P\bigr)$ \label{ln:authdeliver}
  \Comment{the adversary decides when}
\State $(B', \msg') \gets (\V{B},\ \V{m})$ \label{ln:authhonest}
\If{$a \neq \ok$}
  \State $(B', \msg') \gets a$; \ $\V{corr} \gets 1$ \label{ln:authsub}
    \Comment{substituting is allowed, and is recorded}
\EndIf
\Req{$B' = \id.P \ \wedge\ \msg' \in \Msgs$} \label{ln:authaddr}
\State $\V{done} \gets 1$
\State \textbf{return} $(\V{A},\ \msg')$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Leak}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $(\V{A},\ \V{B},\ \V{m},\ \V{corr})$
\algsave
\end{algorithmic}
\end{multicols}}
\end{interface}
