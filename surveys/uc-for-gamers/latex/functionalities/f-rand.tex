%% The interface box of f-rand, the one source for both the book
%% and uc/.../f-rand/index.qmd. Edit here and nowhere else:
%% scripts/gen_interface.py regenerates the page's HTML from this file,
%% and CI fails the build when the two have drifted apart.
\begin{interface}[nofloat]{Functionality $\Frand$ \\
\params{$\PID$, \quad $\Ps$, \quad $\admits$, \quad $\uses := \emptyset$, \quad $\pars := n$}}

{\scriptsize
\setlength{\columnsep}{1.4em}
\begin{multicols}{2}
\opsig{$\op{Initialize}()$}:
\begin{algorithmic}[1]
\setcounter{contline}{0}\algcont
\State $\V{ctr} \gets 0$
\State $\V{L} : \mathbb{N} \to \{0,1\}^{n} \cup \{\unset\}$
\State $\V{L}[*] \gets \unset$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Rnd}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State $\V{ctr} \gets \V{ctr} + 1$
\State $r \gets_{\$} \{0,1\}^{n}$ \label{ln:rndsample}
  \Comment{the one sampling in the paper}
\State $\V{L}[\V{ctr}] \gets r$
\State \textbf{return} $(\V{ctr},\, r)$ \label{ln:rndreturn}
  \Comment{the index, so a caller may erase}
\algsave
\end{algorithmic}
\vspace{1.4ex}

\columnbreak

\opsig{$\id.\op{Erase}(i)$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State $\V{L}[i] \gets \unset$ \label{ln:rnderase}
\State \textbf{return} $\ok$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Leak}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $\V{L}$
\algsave
\end{algorithmic}
\end{multicols}}
\end{interface}
