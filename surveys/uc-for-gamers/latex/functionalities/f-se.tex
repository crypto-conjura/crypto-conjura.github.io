%% The interface box of f-se, the one source for both the book
%% and uc/.../f-se/index.qmd. Edit here and nowhere else:
%% scripts/gen_interface.py regenerates the page's HTML from this file,
%% and CI fails the build when the two have drifted apart.
\begin{interface}[nofloat]{Functionality $\Fse$ \\
\params{$\PID$, \quad $\Ps$, \quad $\admits$, \quad
$\uses := \{(\Adv,\serves)\}$, \quad $\pars := L$}}

{\scriptsize
\setlength{\columnsep}{1.4em}
\begin{multicols}{2}
\opsig{$\op{Initialize}()$}:
\begin{algorithmic}[1]
\setcounter{contline}{0}\algcont
\State $\V{Keys}, \V{Known} \gets \emptyset$
\State $\V{Key} : \Fse.\Ps \times \mathbb{N} \rightharpoonup \V{Keys}$
\State $\V{Next} : \Fse.\Ps \to \mathbb{N}$
\State $\V{Next}[*] \gets 0$
\State $\V{DecTable} : \V{Keys} \to 2^{\{0,1\}^{*} \times \{0,1\}^{*}}$
\State $\V{DecTable}[*] \gets \emptyset$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{KeyGen}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State $(\mathit{corr}, k) \gets \Adv\bigl(\id.\op{KeyGen}\bigr)$
\State $k \gets \San[\Clean_{k}](k; \V{Keys})$
\State $\V{Keys} \gets \V{Keys} \cup \{k\}$
\If{$\mathit{corr} = 1$}
  \State $\V{Known} \gets \V{Known} \cup \{k\}$
\EndIf
\State $\mathit{ptr} \gets \V{Next}[\id.P]$;\ \ $\V{Next}[\id.P] \gets \mathit{ptr} + 1$
\State $\V{Key}[\id.P, \mathit{ptr}] \gets k$
\State \textbf{return} $\mathit{ptr}$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Store}(k)$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\If{$k \in \V{Keys} \setminus \V{Known}$}
  \State \textbf{return} $\none$
    \Comment{a bare guess of an unknown key is refused}
\EndIf
\State $\V{Keys} \gets \V{Keys} \cup \{k\}$;\ \ $\V{Known} \gets \V{Known} \cup \{k\}$
\State $\mathit{ptr} \gets \V{Next}[\id.P]$;\ \ $\V{Next}[\id.P] \gets \mathit{ptr} + 1$
\State $\V{Key}[\id.P, \mathit{ptr}] \gets k$
\State \textbf{return} $\mathit{ptr}$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\columnbreak

\opsig{$\id.\op{Enc}(\mathit{ptr}, \msg)$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State $k \gets \V{Key}[\id.P, \mathit{ptr}]$
\If{$k = \unset$}
  \State \textbf{return} $\none$
\EndIf
\State $c \gets \none$
\If{$k \notin \V{Known}$}
  \State $c \gets \Adv\bigl(\id.\op{Enc}, L(\msg)\bigr)$
  \State $c \gets \San[\Clean_{c}](c)$
  \If{$c \neq \none$}
    \State $\V{DecTable}[k] \gets \V{DecTable}[k] \cup \{(\msg, c)\}$
  \EndIf
\EndIf
\If{$k \in \V{Known}$}
  \State $c \gets \Adv\bigl(\id.\op{Enc}, \msg\bigr)$
\EndIf
\State \textbf{return} $c$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Dec}(\mathit{ptr}, c)$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State $k \gets \V{Key}[\id.P, \mathit{ptr}]$
\If{$k = \unset$}
  \State \textbf{return} $\none$
\EndIf
\State $\msg \gets \none$
\If{$k \notin \V{Known}$}
  \If{$\exists!\, \msg' : (\msg', c) \in \V{DecTable}[k]$}
    \State $\msg \gets \msg'$
      \Comment{unique match; otherwise stays refused}
  \EndIf
\EndIf
\If{$k \in \V{Known}$}
  \State $\msg \gets \Adv\bigl(\id.\op{Dec}, c\bigr)$
\EndIf
\State \textbf{return} $\msg$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Leak}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $\bigl(\{(\mathit{ptr}, \V{Key}[\id.P, \mathit{ptr}])\}_{\mathit{ptr} < \V{Next}[\id.P]},$
\Statex \quad\ $\{(k, \V{DecTable}[k]) : \exists\, \mathit{ptr}\, .\ \V{Key}[\id.P,\mathit{ptr}] = k\}\bigr)$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\Clean_{k}(k; \V{Keys})$}:
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $k \in \{0,1\}^{*} \ \wedge\ k \notin \V{Keys}$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\Clean_{c}(c)$}:
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $c \in \{0,1\}^{*}$
\algsave
\end{algorithmic}
\end{multicols}}
\end{interface}
