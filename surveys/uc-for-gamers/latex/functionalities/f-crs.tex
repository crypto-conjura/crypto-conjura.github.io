\begin{interface}[nofloat]{Functionality $\Fcrs$ \\
\params{$\PID$, \quad $\Ps$, \quad $\admits$, \quad $\uses := \emptyset$, \quad $\pars := D$}}

{\scriptsize
\setlength{\columnsep}{1.4em}
\begin{multicols}{2}
\opsig{$\op{Initialize}()$}:
\begin{algorithmic}[1]
\setcounter{contline}{0}\algcont
\State $\V{crs} \gets \unset$
\algsave
\end{algorithmic}
\vspace{1.4ex}

\opsig{$\id.\op{Get}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\If{$\V{crs} = \unset$}
  \State $\V{crs} \gets_{\$} D$ \label{ln:crssample}
    \Comment{sampled once, on the first ask}
\EndIf
\State \textbf{return} $\V{crs}$ \label{ln:crssame}
\algsave
\end{algorithmic}
\vspace{1.4ex}

\columnbreak

\opsig{$\id.\op{Leak}()$} \quad from $\id'$
\begin{algorithmic}[1]
\algcont
\State \textbf{return} $\V{crs}$ \label{ln:crsleak}
  \Comment{$\unset$ until a party asks}
\algsave
\end{algorithmic}
\end{multicols}}
\end{interface}
